## 計算式（実装の中心部分・JavaScript）
```js
// pip: USDJPY 0.01 / EURUSD 0.0001 / GOLD 0.1, cs: FX 100000 / GOLD 100
// conv: USDJPY=1, EURUSD・GOLD=USDJPYレート, commConv: 手数料(USD)の円換算
function pl(c,bid){
  var p=c.pip, sp0=c.sp0*p, sp1=c.sp1*p, slip=c.slip*p;
  var d=(bid+sp1/2)-(c.b0+sp0/2), half=(sp0+sp1)/2;   // 仲値の変化／各ポジションのスプレッド負担
  function side(lot,dir,sw){
    var k=lot*c.cs*c.conv;
    return {p:dir*d*k, s:-half*k, c:-c.comm*lot*c.commConv + sw*lot*c.days, l:-slip*k};
  }
  var B=side(c.lb,1,c.swB), S=side(c.ls,-1,c.swS);
  B.t=B.p+B.s+B.c+B.l; S.t=S.p+S.s+S.c+S.l;
  return {B:B,S:S,tot:B.t+S.t};
}
// 検算：買い損益＝(現在Bid−開始Ask)、売り損益＝(開始Bid−現在Ask) と一致（agentCL確認）

// 証拠金
mB=lotB*cs*price円/lev; mS=lotS*cs*price円/lev;
両建て中=|mB−mS|;  フルロット=残高*lev/(cs*price円)

// 小さい方を決済した後
closeBuy = B.t < S.t;  Lnet=決済側の損益; Wnet=残す側の損益
per = 残す側lot*cs*conv           // 価格1.0動いた時の円
目標までの値動き = (目標 − Lnet − Wnet) / per
有効証拠金 eq = 残高 + Lnet + Wnet;  維持率 = eq / 残す側の証拠金
ロスカットまでの逆行 = (eq − 0.2×証拠金) / per   // マージンコールは0.5

// 比率（伸びている側：小さい側）＝ |小さい側| ÷ 伸びている側（伸びている側がプラスの時のみ）
```
